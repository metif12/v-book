Control Flow

If

fn main() {
    age := 25
    if age >= 18 {
        println('Adult')
    } else {
        println('Minor')
    }
}

If as expression

fn main() {
    age := 25
    category := if age >= 18 { 'adult' } else { 'minor' }
    println(category)
}

For loop

fn main() {
    // Over an array
    fruits := ['apple', 'banana', 'cherry']
    for fruit in fruits {
        println(fruit)
    }

    // Range
    for i in 0 .. 5 {
        println(i)
    }

    // With index
    for i, fruit in fruits {
        println('${i}: ${fruit}')
    }
}

Match

fn main() {
    color := 'blue'
    match color {
        'red' { println('Red!') }
        'blue' { println('Blue!') }
        else { println('Other color') }
    }
}
